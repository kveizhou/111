import json, re, sys, tempfile
from pathlib import Path
from types import SimpleNamespace
sys.path.insert(0, "server/site")
import collect_cascode_batch as cb

session, read, preflight, batch = map(Path, sys.argv[1:])
bound = cb.inputs(SimpleNamespace(session=session, maestro_read=read, preflight=preflight))
if not batch.is_absolute() or not (batch / "status.json").is_file():
    sys.exit("STOP: batch output directory is missing")
out = Path(tempfile.mkdtemp(prefix="dc_review_ports_", dir=str(batch)))
selection = bound["data"]["selection"]
values = [selection[k] for k in ("bench_library", "bench_cell", "setup_view")]
summary = ["CHECKER=cascode-dc-evidence-v2", "SPECTRE_RERUN=false",
           "SELECTED_CORNER=" + bound["settings"]["corner"],
           "READ_SCOPE=saved_base_test_settings"]
code = 'load(getShellEnvVar("ST_SKILL_INIT"))\n'
code += 'let((stReviewSetup stReviewTest stReviewResult stReviewOceanPort stReviewDonePort)\n'
code += 'stReviewResult=errset(prog( ()\n'
code += 'stReviewSetup=maeOpenSetup(' + ' '.join(json.dumps(v) for v in values) + ' ?mode "r")\n'
code += 'unless(stReviewSetup error("Cannot open saved setup read-only"))\n'
code += 'stReviewTest=maeGetTestSession(' + json.dumps(bound["settings"]["test_name"]) + ' ?session stReviewSetup)\n'
code += 'unless(stReviewTest error("Cannot read saved test"))\n'
code += 'stReviewOceanPort=asiOpenOceanScript(getShellEnvVar("ST_REVIEW_OCEAN"))\n'
code += 'unless(stReviewOceanPort error("Cannot open OCEAN export port"))\n'
code += 'unless(asiWriteOceanScript(stReviewOceanPort stReviewTest ?noRun t ?fullKey t)\n'
code += 'error("Cannot export saved test settings"))\n'
code += 'asiCloseOceanScript(stReviewOceanPort) stReviewOceanPort=nil\n'
code += 'stReviewDonePort=outfile(getShellEnvVar("ST_REVIEW_DONE"))\n'
code += 'unless(stReviewDonePort error("Cannot write completion marker"))\n'
code += 'fprintf(stReviewDonePort "OK\\n") close(stReviewDonePort) stReviewDonePort=nil t\n) t)\n'
code += 'when(stReviewOceanPort errset(asiCloseOceanScript(stReviewOceanPort) t))\n'
code += 'when(stReviewDonePort close(stReviewDonePort))\n'
code += 'unless(and(stReviewResult car(stReviewResult)) printf("ST_REVIEW_FAILED\\n"))\n)\nexit()\n'
(out / "read.il").write_text(code)
(out / "cds.lib").write_text("INCLUDE " + json.dumps(str(bound["cds"])) + "\n")
saved = out / "saved_test.ocn"
env = dict(bound["env"], ST_REVIEW_OCEAN=str(saved), ST_REVIEW_DONE=str(out / "done"))
count = 0
try:
    cb.command([cb.executable(env, "ST_VIRTUOSO_BIN", "virtuoso"), "-nograph", "-replay",
                str(out / "read.il"), "-log", str(out / "virtuoso.log")], out, env, 180, out / "launcher.log")
    if not saved.is_file() or not saved.stat().st_size or not (out / "done").is_file():
        raise ValueError("saved test export is incomplete")
    if (out / "done").read_text().strip() != "OK":
        raise ValueError("saved test completion marker is invalid")
    source = saved.read_text()
    forms = []
    for hit in re.finditer(r"(?m)^\s*(analysis|option|envOption|temp|ic|nodeset)\s*\(", source):
        start = source.index("(", hit.start())
        depth = 0; quoted = False; escaped = False; comment = False
        for pos in range(start, len(source)):
            char = source[pos]
            if comment:
                if char == "\n": comment = False
            elif quoted:
                if escaped: escaped = False
                elif char == "\\": escaped = True
                elif char == '"': quoted = False
            elif char == ";": comment = True
            elif char == '"': quoted = True
            elif char == "(": depth += 1
            elif char == ")":
                depth -= 1
                if depth == 0: break
        else:
            raise ValueError("unclosed saved setting")
        block = source[hit.start():pos + 1].strip()
        forms.append(block)
        flat = " ".join(block.split())
        for offset in range(0, len(flat), 1000):
            prefix = "SAVED_TEST " if offset == 0 else "SAVED_TEST_CONT "
            summary.append(prefix + flat[offset:offset + 1000])
    if not forms: raise ValueError("no saved analysis/options found in export")
    cb.unchanged(bound)
    (out / "settings_forms.json").write_text(json.dumps(forms, ensure_ascii=False, indent=2) + "\n")
    count = len(forms)
    summary += ["SAVED_FORM_COUNT=" + str(count), "SAVED_TEST_SCRIPT_READ_OK",
                "SAVED_SETTINGS_CAPTURED_NOT_APPLIED=true", "FIXED_INPUTS_UNCHANGED=true"]
except (OSError, ValueError) as error:
    summary.append("SAVED_TEST_READ_INCOMPLETE=" + str(error))
    summary += ["NATIVE_ERROR " + item for item in cb.brief_errors(out)]
shown = summary[:170]
shown += ["SUMMARY_OMITTED_LINES=" + str(max(0, len(summary) - 170)),
          "ELECTRICAL_VERIFIED=false", "VBIAS_GENERATED=false", "RAW_DETAILS_REMAIN_ON_SERVER=true"]
(out / "summary.txt").write_text("\n".join(shown) + "\n")
print("\n".join(shown))
sys.exit(0 if count > 0 else 2)
'PY'

3. 成功时预期显示
开头：
CHECKER=cascode-dc-evidence-v2
SPECTRE_RERUN=false
SELECTED_CORNER=Copy_PT_FUNC_703
READ_SCOPE=saved_base_test_settings
中间出现实际保存的设置，例如：
SAVED_TEST analysis(...)
SAVED_TEST option(...)
SAVED_TEST envOption(...)
长设置会以 SAVED_TEST_CONT 接续显示。
末尾必须包含：
SAVED_TEST_SCRIPT_READ_OK
SAVED_SETTINGS_CAPTURED_NOT_APPLIED=true
FIXED_INPUTS_UNCHANGED=true
ELECTRICAL_VERIFIED=false
VBIAS_GENERATED=false
RAW_DETAILS_REMAIN_ON_SERVER=true
求解设置检查退出码=0
这里读取的是基准测试保存的设置；即使显示 temp(27)，也不代表把已选 Corner 的 125°C 改成了常温。
执行完把这次完整摘要和退出码贴回来。若出现 SAVED_TEST_READ_INCOMPLETE 或退出码非零，就停在这里。 读取成功后才能根据实际设置判断下一次 DC 应如何运行；目前还不能生成已验证的真实 VBIAS。
