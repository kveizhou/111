import json, sys
from pathlib import Path

data = json.loads(Path(sys.argv[1]).read_text())
if data.get("result") != "READ_OK":
    raise SystemExit("STOP: 原配置读取未成功")

rows = data["records"]
global_names = {r["name"] for r in rows if r["kind"] == "global_variable"}

def value(kind, scope, name):
    hits = [r for r in rows if (r["kind"], r["scope"], r["name"]) == (kind, scope, name)]
    return hits[0]["raw_value"] if len(hits) == 1 and hits[0]["status"] == "OK" else "UNKNOWN"

corners = [r for r in rows if r["kind"] == "corner" and r["name"] == "enabled"]
active = [r["scope"] for r in corners if r["raw_value"].strip() == "t" and r["status"] == "OK"]

lines = [
    "READER=maestro-focus-v1",
    "ENABLED_CORNERS=" + (",".join(active) if active else "(none)"),
    "CORNER_FLAGS=" + " ".join(r["scope"] + "=" + r["raw_value"] for r in corners)
]

for r in rows:
    if r["kind"] == "test" and r["name"] in ("test_name", "design_view", "models", "enabled", "all_global_variables_disabled"):
        lines.append("TEST [" + r["scope"] + "] " + r["name"] + "=" + r["raw_value"])

for corner in active:
    for r in rows:
        if r["scope"] == corner and r["kind"] == "corner_variable":
            lines.append("CORNER_VARIABLE [" + corner + "] " + r["name"] + "=" + r["raw_value"] +
                         " ENABLED=" + value("corner_variable_enabled", corner, r["name"]))
        if r["scope"] == corner and r["kind"] == "corner_model_file":
            enabled = value("corner_model_enabled", corner, r["name"])
            if enabled.strip() != "nil":
                lines.append("CORNER_MODEL [" + corner + "] " + r["name"] +
                             " FILE=" + r["raw_value"] +
                             " SECTION=" + value("corner_model_section", corner, r["name"]) +
                             " ENABLED=" + enabled)

for r in rows:
    if r["kind"] == "global_variable":
        lines.append("GLOBAL_VARIABLE " + r["name"] + "=" + r["raw_value"] +
                     " ENABLED=" + value("global_variable_enabled", r["scope"], r["name"]))
    if r["kind"] == "test_variable":
        flag = value("test_variable_global_override_enabled", r["scope"], r["name"]) if r["name"] in global_names else "N/A_LOCAL_VARIABLE"
        lines.append("TEST_VARIABLE [" + r["scope"] + "] " + r["name"] +
                     "=" + r["raw_value"] + " GLOBAL_OVERRIDE=" + flag)

for line in lines[:180]:
    print(" ".join(line.splitlines()))

print("FOCUS_OMITTED_LINES=" + str(max(0, len(lines) - 180)))
print("FOCUSED_CONFIG_READ_COMPLETE")
